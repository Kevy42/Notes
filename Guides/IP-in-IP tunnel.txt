Prerequisite context:

    # example@server-1:~$ ip -c -br a
    # lo               UNKNOWN        127.0.0.1/8 ::1/128
    # ens18            UP             10.0.0.10/24 fe80::be24:11ff:fee2:c5d3/64

    # example@server-2:~$ ip -c -br a
    # lo               UNKNOWN        127.0.0.1/8 ::1/128
    # ens18            UP             10.0.0.20/24 fe80::be24:11ff:fee2:c5d3/64


1. Enable IP forwarding
    # Not strictly needed, but good to have
    sudo sysctl net.ipv4.ip_forward=1 # On server-1
    sudo sysctl net.ipv4.ip_forward=1 # On server-2

    # Check to make sure it's enabled
    sudo sysctl --value net.ipv4.ip_forward # On server-1 .Should return "1"
    sudo sysctl --value net.ipv4.ip_forward # On server-2 .Should return "1"


2. Configure Foo-over-UDP (FOU)/Generic UDP Encapsulation (GUE) receive endpoints (skip if using native IP-in-IP)
    sudo modprobe fou # On server-1
    sudo modprobe fou # On server-2

    # IP-in-IP with Foo-over-UDP (FOU) encapsulation
        # "ipproto PROTOCOL-NUMBER" tells the kernel what protocol the received packets should be treated has having given that
        # FOU encapsulation doesn't add any protocol identification header (unlike GUE)
        sudo ip fou add port 5000 ipproto 4 local 10.0.0.10 dev ens18 # On server-1
        sudo ip fou add port 5000 ipproto 4 local 10.0.0.20 dev ens18 # On server-2

    # IP-in-IP with Generic UDP Encapsulation (GUE)
        sudo ip fou add port 5000 gue local 10.0.0.10 dev ens18 # On server-1
        sudo ip fou add port 5000 gue local 10.0.0.20 dev ens18 # On server-2

    # Check to make sure the FOU/GUE entry have been added
    # example@server-1:~$ ip fou show
    # port 5000 gue local 10.0.0.10 dev ens18

    # example@server-2:~$ ip fou show
    # port 5000 gue local 10.0.0.20 dev ens18


3. Set up the IP-in-IP tunnels (see ip-link(8) for full syntax).
    # Native IP-in-IP
        # "remote" and "local" should be set to the servers routable IPs respectively as they'll be used for the packets outer IP —
        # the underlay IP
        sudo ip link add ipip0 type ipip remote 10.0.0.20 local 10.0.0.10 encap none mode ipip # On server-1
        sudo ip link add ipip0 type ipip remote 10.0.0.10 local 10.0.0.20 encap none mode ipip # On server-2

    # IP-in-IP with Foo-over-UDP (FOU) encapsulation.
        sudo ip link add ipip0 type ipip remote 10.0.0.20 local 10.0.0.10 encap fou encap-dport 5000 mode ipip # On server-1
        sudo ip link add ipip0 type ipip remote 10.0.0.10 local 10.0.0.20 encap fou encap-dport 5000 mode ipip # On server-2

    # IP-in-IP with Generic UDP Encapsulation (GUE)
        sudo ip link add ipip0 type ipip remote 10.0.0.20 local 10.0.0.10 encap gue encap-dport 5000 mode ipip # On server-2
        sudo ip link add ipip0 type ipip remote 10.0.0.10 local 10.0.0.20 encap gue encap-dport 5000 mode ipip # On server-2

    # Check to make sure the tunnel interfaces have been added
    # example@server-1:~$ ip -c address show dev ipip0
    # 4: ipip0@NONE: <POINTOPOINT,NOARP> mtu 1468 qdisc noop state DOWN group default qlen 1000
    #     link/ipip 10.0.0.10 peer 10.0.0.20
    #     inet 172.16.0.1 peer 172.16.0.2/32 scope link ipip0
    #        valid_lft forever preferred_lft forever

    # example@server-1:~$ ip -c address show dev ipip0
    # 4: ipip0@NONE: <POINTOPOINT,NOARP> mtu 1468 qdisc noop state DOWN group default qlen 1000
    #     link/ipip 10.0.0.10 peer 10.0.0.20
    #     inet 172.16.0.1 peer 172.16.0.2/32 scope link ipip0
    #        valid_lft forever preferred_lft forever


3. Add IPs to the interfaces
    # These will be added to the packets inner IP; the overlay IP
    sudo ip address add local 172.16.0.1 peer 172.16.0.2/32 scope link dev ipip0 # On server-1
    sudo ip address add local 172.16.0.2 peer 172.16.0.1/32 scope link dev ipip0 # On server-2


4. Bring up the interfaces
    sudo ip link set up dev ipip0 # On server-1
    sudo ip link set up dev ipip0 # On server-2

    # Check to make sure the routes have been added
    # example@server-1:~$ ip -c route
    # default via 10.0.0.1 dev ens18
    # 10.0.0.0/24 dev ens18 proto kernel scope link src 10.0.0.10
    # 172.16.0.2 dev ipip0 proto kernel scope link src 172.16.0.1

    # example@server-2:~$ ip -c route
    # default via 10.0.0.1 dev ens18
    # 10.0.0.0/24 dev ens18 proto kernel scope link src 10.0.0.20
    # 172.16.0.1 dev ipip0 proto kernel scope link src 172.16.0.2


5. Test the connection
    # example@server-1:~$ ping 172.16.0.2
    # PING 172.16.0.2 (172.16.0.2) 56(84) bytes of data.
    # 64 bytes from 172.16.0.2: icmp_seq=1 ttl=64 time=0.685 ms
    # 64 bytes from 172.16.0.2: icmp_seq=2 ttl=64 time=0.358 ms

    # On server-2. "proto 4" is ipip.
    # example@server-2:~$ sudo tcpdump -e --interface any -n --print --direction inout -q -tttt --packet-buffered -w "/tmp/tcpdump_$(printf '%(%Y-%m-%d_%H-%M-%S)T' -1).pcap" proto 4
    # tcpdump: listening on any, link-type LINUX_SLL2 (Linux cooked v2), snapshot length 262144 bytes
    # 2026-09-24 14:23:30.308272 ens18 In  ifindex 2 bc:24:11:e2:c5:d3 10.0.0.10 > 10.0.0.20: 172.16.0.1 > 172.16.0.2: ICMP echo request, id 11, seq 6, length 64
    # 2026-09-24 14:23:30.308304 ens18 Out ifindex 2 bc:24:11:b0:c4:bc 10.0.0.20 > 10.0.0.10: 172.16.0.2 > 172.16.0.1: ICMP echo reply, id 11, seq 6, length 64


6. Cleanup once done
    # Disable IP forwarding if it already was disabled prior to enabling in step 1
    sudo sysctl net.ipv4.ip_forward=0 # On server-1
    sudo sysctl net.ipv4.ip_forward=0 # On server-2

    sudo ip link del dev ipip0 # On server-1
    sudo ip link del dev ipip0 # On server-2

    # Remove the FOU/GUE receive endpoints
    sudo ip fou del port 5000 local 10.0.0.10 dev ens18 # On server-1
    sudo ip fou del port 5000 local 10.0.0.20 dev ens18 # On server-2

    sudo modprobe --remove ipip fou # On server-1
    sudo modprobe --remove ipip fou # On server-2
