modprobe ipip # NEEDED?

1. Set up IPIP tunnel (see ip-link(8) for full syntax).

    sudo ip link add ipip0 type ipip remote EXISTING-INTERFACE-IP-ON-REMOTE local EXISTING-INTERFACE-IP encap none mode ipip # On server-1
    sudo ip link add ipip0 type ipip remote EXISTING-INTERFACE-IP-ON-REMOTE local EXISTING-INTERFACE-IP encap none mode ipip # On server-2

    # "EXISTING-INTERFACE-IP" and "EXISTING-INTERFACE-IP-ON-REMOTE" should be substituted for the servers routable IPs (see example below)
    # respectively; It'll be used as the packet's outer IP.

    # example@server-1:~$ ip -c -br a
    # lo               UNKNOWN        127.0.0.1/8 ::1/128
    # ens18            UP             10.0.0.10/24 fe80::be24:11ff:fee2:c5d3/64 <------ "EXISTING-INTERFACE-IP"/"EXISTING-INTERFACE-IP-ON-REMOTE" on server-1/server-2

2. Add IPs to the interfaces
    sudo ip address add local 172.0.0.1/30 broadcast - dev ipip0 # On server-1
    sudo ip address add local 172.0.0.2/30 broadcast - dev ipip0 # On server-2

    # Check to make sure the routes have been added
    ip -c route

    # example@server-1:~$ ip -c route
    # default via 10.0.0.1 dev ens18
    # 10.0.0.0/24 dev ens18 proto kernel scope link src 10.0.0.10
    # 172.0.0.0/30 dev ipip0 proto kernel scope link src 172.0.0.1

3. Bring the interface up
    sudo ip link set up dev ipip0 # On server-1
    sudo ip link set up dev ipip0 # On server-2

4. Test
    ping 172.0.0.2

    # On server-2. "proto 4" is ipip.
    sudo tcpdump -e --interface any -n --print --direction inout -q -tttt --packet-buffered -w "/tmp/tcpdump_$(printf '%(%Y-%m-%d_%H-%M-%S)T' -1).pcap" proto 4
