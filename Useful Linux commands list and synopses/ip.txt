# Save, view and restore IP state
ip address save > ip-backup.txt # Can be limited to one interface
ip address showdump < ip-backup.txt
ip address restore < ip-backup.txt

# Add interface. See ip-link(8) for full syntax
ip link { add | del } \
    [INTERFACE] \
    name [INTERFACE NAME] \  # Sometimes required
    type [TYPE] \
    [TYPE SPECIFIC OPTIONS]

# Add interface that's physically linked to another (e.g., of type "vlan"). See ip-link(8) for full syntax
ip link { add | del } \
    link \
    [PHYSICAL INTERFACE] \   # Physical interface to link to
    name [INTERFACE NAME] \  # Interface to be created
    type [TYPE] \
    [TYPE SPECIFIC OPTIONS]


# Change interface settings
ip link { set | change } \
    dev [INTERFACE] \
    [SETTING] { on | off | STRING | NUMBER } \  # Relevant settigns: "arp", "multicast", "allmulticast", "promisc", "name", "mtu", "address", "broadcast", "alias", "master", "nomaster", "addrgenmode". "nomaster" takes no arguments.

# Change interface type specific settings. See ip-link(8) for full syntax
ip link { set | change } \
    dev [INTERFACE] \
    type [TYPE] \
    [TYPE SPECIFIC OPTIONS]


# Add IP to interface
ip address { add | del | change | replace } \
    local [ADDRESS/PREFIX] \
    broadcast [BROADCAST ADDRESS] \  # Optional. Can be set to "+" or "-" in order to derive it from the local [ADDRESS/PREFIX]
    scope [SCOPE] \                  # Optinal. Affects address selection and routing, can be "global", "link" or "host". "link" is only accessible on this layer 2 segment (e.g., 169.254.0.0/16) and "host" only this machine (i.e., loopback)
    dev [INTERFACE] \
    valid_lft [LFT] \                # Optinal. How long the IP is valid for
    preferred_lft [LFT]              # Optinal. How long before the IP is considered "deprecated" and ceases to be used for new outgoing connections

# Bring interface up/down
ip link set dev [INTERFACE] { up | down }


ip route get \
    [DST IP]
    from [SRC IP] \            # Optional. Packet source IP, must be available on one of the interfaces unless "iif" is also set
    iif [INTERFACE] \          # Optional. Check if packet can be routed (forwarded) if it arrived on this interface with a destination IP specified by "from"
    mark [MARK] \              # Optional
    ipproto [PROTOCOL NAME] \  # Optional. Filter based on protocol
    sport [PORT] \             # Optional. Filter based on source port
    dport [PORT]               # Optional. Filter based on destination port

# Filter for routing table entries (as opposed to what route is chosen with "ip route get")
ip route show \
    [FILTER] \            # Optional. "root", "match" or "exact". See ip-route(8) for details
    [IP/PREFIX] \         # Optional
    table [NAME or ID] \  # Optional. "all" can be used to match on all tables
    protocol [STRING] \   # Optional. Filter based on "proto" type, e.g, "static"
    scope [STRING] \      # Optional. Filter based on "scope" type, e.g., "link"
    type [TYPE] \         # Optional. Filter based on "type" type, e.g., "unicast"
    dev [INTERFACE] \     # Optional
    via [IP/PREFIX] \     # Optional


ip route { add | del | change | append | replace } \
    [TYPE] \              # Optional. Can be "unicast" (default), "local", "broadcast", "multicast", "throw", "unreachable", "prohibit" or "blackhole". "local" and "broadcast" will be added to the "local" rule table \
    [IP/PREFIX] \
    table [NAME or ID] \  # Optional
    proto [STRING] \      # Optional. Can be "kernel", "boot" or "static" (other options are irrelevant). "kernel" is added by the kernel, "boot" during boot and "static" manually
    scope [STRING] \      # Optional. Can be "link", "host" or "global". "host" is only accessible on this machine, "link" only on this layer 2 segment and "global" from anywhere
    metric [NUMBER] \     # Optional. Defaults to 0, lower takes precedence
    via [IP/PREFIX] \     # Optional
    dev [INTERFACE] \     # Optional
    onlink \              # Optional. Consider the route valid even if the next hop is not reachable via the interface
    mtu lock [NUMBER] \   # Optional. Either with or without "lock" which inhibits PMTUD from updating the MTU
    advmss [NUMBER]       # Optional. Advertised MSS


ip rule show \
    not \                        # Optional. Invert filter
    from [IP/PREFIX] \           # Optional
    to [IP/PREFIX] \             # Optional
    fwmark [MARK] \              # Optional
    iif [INTERFACE] \            # Optional. In interface
    oif [INTERFACE] \            # Optional. Out interface
    priority [NUMBER] \          # Optional
    ipproto [PROTOCOL] \         # Optional
    sport [PORT | PORT RANGE] \  # Optional. Source port
    dport [PORT | PORT RANGE] \  # Optional. Destination port

ip rule { add | del } \
    not \                           # Optional. Invert match criteria
    from [IP/PREFIX] \              # Optional
    to [IP/PREFIX] \                # Optional
    fwmark [MARK] \                 # Optional
    iif [INTERFACE] \               # Optional. In interface
    oif [INTERFACE] \               # Optional. Out interface
    priority [NUMBER] \             # Optional but should be set explicitly. Lower numbers take precedence
    ipproto [PROTOCOL] \            # Optional
    sport [PORT | PORT RANGE] \     # Optional. Source port
    dport [PORT | PORT RANGE] \     # Optional. Destination port

    NEEDED? [type] [TYPE] \                   # Optional. Can be "unicast" (default), "blackhole", "unreachable", "prohibit", "nat"

    table [NAME or ID] \
    protocol [PROTO] \              # Optional. Can be "kernel", "boot" or "static" (other options are irrelevant). "kernel" is added by the kernel, "boot" during boot and "static" manually
    goto [NUMBER] \                 # Optional. Jump to table with priority [NUMBER] and continue matching from there. Does *not* enter the table, just jumps to it and evaluates from there like normal

    suppress_prefixlength [NUMBER]  # Optional. Reject *routes* with prefix [NUMBER] or less in the table

    - View route "proto" types /usr/share/iproute2/rt_protos
